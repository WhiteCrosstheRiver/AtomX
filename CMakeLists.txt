cmake_minimum_required(VERSION 3.24)
project(AtomX VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "The current AtomX UI renderer requires Windows and Direct3D 11")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui")
add_library(imgui STATIC ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp ${IMGUI_DIR}/backends/imgui_impl_win32.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC ${IMGUI_DIR})
add_executable(AtomX WIN32 src/main.cpp)
target_link_libraries(AtomX PRIVATE imgui d3d11 dxgi d3dcompiler windowscodecs ole32 comdlg32 shell32 dwmapi)
if(MSVC)
 target_compile_options(AtomX PRIVATE /W4 /utf-8 /EHsc)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
add_test(NAME core COMMAND core_tests)
add_executable(atomx_benchmark tests/benchmark.cpp)
