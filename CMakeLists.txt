cmake_minimum_required(VERSION 3.24)
project(AtomX VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "The current AtomX UI renderer requires Windows and Direct3D 11")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui")
add_library(imgui STATIC ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp ${IMGUI_DIR}/backends/imgui_impl_win32.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC ${IMGUI_DIR})
enable_language(RC)
add_executable(AtomX WIN32 src/main.cpp src/atomx.rc)
target_link_libraries(AtomX PRIVATE imgui d3d11 dxgi d3dcompiler windowscodecs ole32 comdlg32 shell32 dwmapi)
if(MSVC)
 target_compile_options(AtomX PRIVATE /W4 /utf-8 /EHsc)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
add_test(NAME core COMMAND core_tests)
add_executable(poscar_tests tests/poscar_tests.cpp)
add_test(NAME poscar COMMAND poscar_tests)
add_executable(structure_io_tests tests/structure_io_tests.cpp)
add_test(NAME structure_io COMMAND structure_io_tests)
add_executable(render_shapes tests/render_shapes.cpp)
target_link_libraries(render_shapes PRIVATE d3d11 dxgi d3dcompiler windowscodecs ole32)
add_test(NAME render_shapes COMMAND render_shapes)
add_executable(export_workflow tests/export_workflow.cpp src/atomx.rc)
target_link_libraries(export_workflow PRIVATE imgui d3d11 dxgi d3dcompiler windowscodecs ole32 comdlg32 shell32 dwmapi)
add_test(NAME export_workflow COMMAND export_workflow)
add_executable(atomx_benchmark tests/benchmark.cpp)
if(MSVC)
 foreach(test_target core_tests poscar_tests structure_io_tests render_shapes export_workflow)
  target_compile_options(${test_target} PRIVATE /W4 /utf-8 /EHsc)
 endforeach()
endif()
